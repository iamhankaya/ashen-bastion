# Denetim yakalama: render'lı koşu, ekran görüntüleri + dokunma hedefi dökümü.
var OUT := "/tmp/claude-0/-home-user-ashen-bastion/8f0bc3f8-a83e-59b3-aca5-b8430a791030/scratchpad/out/"
var c
var log_lines: Array = []

func _initialize() -> void:
	_run()

func shot(n: String) -> void:
	await process_frame
	await RenderingServer.frame_post_draw
	var img := root.get_texture().get_image()
	img.save_png(OUT + n + ".png")
	log_lines.append("shot %s %dx%d" % [n, img.get_width(), img.get_height()])

func frames(n: int) -> void:
	for i in n:
		await process_frame

func secs(s: float) -> void:
	var t := Time.get_ticks_msec()
	while Time.get_ticks_msec() - t < s * 1000.0:
		await process_frame

func rects(tag: String, node: Node) -> void:
	var out: Array = []
	_collect(node, out)
	log_lines.append("RECTS %s %s" % [tag, JSON.stringify(out)])

func _collect(n: Node, out: Array) -> void:
	if n is Control and n.is_visible_in_tree():
		if n is Button or n is Label:
			var r: Rect2 = n.get_global_rect()
			out.append([n.get_class(), n.name, (n.text if "text" in n else ""), snappedf(r.position.x,1), snappedf(r.position.y,1), snappedf(r.size.x,1), snappedf(r.size.y,1), (n.get_theme_font_size("font_size") if n is Label or n is Button else 0)])
	for ch in n.get_children():
		_collect(ch, out)

func touch(idx: int, pressed: bool, pos: Vector2) -> void:
	var e := InputEventScreenTouch.new()
	e.index = idx
	e.pressed = pressed
	e.position = pos
	Input.parse_input_event(e)

func drag(idx: int, pos: Vector2, rel := Vector2.ZERO) -> void:
	var e := InputEventScreenDrag.new()
	e.index = idx
	e.position = pos
	e.relative = rel
	Input.parse_input_event(e)

func clear_levelups() -> int:
	var n := 0
	var guard := 0
	while c.screen == 2 and guard < 40:
		guard += 1
		await secs(0.7)
		if c.level_up.cards.is_empty():
			break
		var cc: Vector2 = c.level_up.cards[0].get_global_rect().get_center()
		touch(0, true, cc)
		await frames(2)
		touch(0, false, cc)
		await frames(4)
		n += 1
	await secs(0.8)
	return n

func _run() -> void:
	var scene: PackedScene = load("res://scenes/boot.tscn")
	c = scene.instantiate()
	root.add_child(c)
	await secs(1.5)
	log_lines.append("viewport %s window %s" % [root.get_visible_rect().size, DisplayServer.window_get_size()])
	await shot("01-menu")
	rects("menu", c.menu)
	# --- touch-only düğme testi (emulate_mouse_from_touch=false)
	var played := [false]
	c.menu.play_requested.connect(func(): played[0] = true)
	var pb: Button = c.menu.play_button
	var center: Vector2 = pb.get_global_rect().get_center()
	touch(0, true, center)
	await frames(3)
	touch(0, false, center)
	await frames(5)
	log_lines.append("TOUCH_ONLY_PLAY_BUTTON fired=%s mouse_emul=%s" % [played[0], ProjectSettings.get_setting("input_devices/pointing/emulate_mouse_from_touch")])
	if not played[0]:
		c.start_run()
	await secs(1.0)
	c.sim.hp = c.sim.max_hp
	await shot("02-hud-start")
	rects("hud", c.hud)
	# kalabalık
	c.sim.debug_fill(&"curumus_piyade", 60, 620.0)
	c.sim.debug_fill(&"les_kargasi", 40, 800.0)
	c.sim.debug_fill(&"sovalye_hayaleti", 6, 480.0)
	await secs(2.0)
	await shot("03-hud-kalabalik")
	# doldurma yakala
	var got := false
	for i in 400:
		await process_frame
		if c.sim.magazine.reloading and c.sim.magazine.reload_total > 0.0 and c.sim.magazine.reload_left < c.sim.magazine.reload_total * 0.5:
			got = true
			break
	log_lines.append("reload_caught=%s" % got)
	await shot("04-hud-doldurma")
	# yuvalar dolu
	for id in [&"seken_mermi", &"son_mermi_patlar", &"cift_tabanca"]:
		c.sim.debug_grant(id, 3)
	for id in [&"barut_kesesi", &"hizli_doldurma", &"fazladan_mermi", &"agir_kursun", &"cevik_cizme", &"saglam_yelek", &"miknatis", &"hizli_parmak"]:
		c.sim.debug_grant(id, 2)
	await secs(0.5)
	await shot("05-hud-yuvalar")
	# düşük can
	c.sim.hp = c.sim.max_hp * 0.2
	await secs(0.4)
	await shot("06-hud-dusuk-can")
	c.sim.hp = c.sim.max_hp
	c.hud.show_boss_band(&"boss")
	await frames(3)
	await shot("07-hud-boss-bandi")
	c.hud.show_boss_band(&"lord")
	await frames(3)
	await shot("07b-hud-lord-bandi")
	log_lines.append('pre_joystick_levelups_cleared=%d screen=%d' % [await clear_levelups(), c.screen])
	c.sim.hp = c.sim.max_hp
	# joystick (yalnız dokunma)
	var o := Vector2(540, 1700)
	touch(0, true, o)
	drag(0, o + Vector2(120, -60), Vector2(120, -60))
	await secs(0.5)
	log_lines.append("move_vec=%s player=%s" % [c.gate.move(), c.sim.player_pos])
	await shot("08-joystick")
	# ikinci parmak
	touch(1, true, Vector2(200, 1900))
	drag(1, Vector2(300, 1900))
	await frames(5)
	log_lines.append("second_finger_move=%s" % c.gate.move())
	touch(1, false, Vector2(300, 1900))
	touch(0, false, o)
	await frames(3)
	# seviye atlama
	c.sim.debug_add_xp(c.sim.xp_to_next - c.sim.xp + 1)
	await secs(0.3)
	log_lines.append("screen=%s lvl_locked=%s" % [c.screen, c.level_up.is_locked()])
	await shot("09-seviye-atlama-kilitli")
	await secs(0.8)
	await shot("10-seviye-atlama")
	rects("levelup", c.level_up)
	# kart seçimini dokunmayla dene
	var cardc: Vector2 = c.level_up.cards[1].get_global_rect().get_center()
	touch(0, true, cardc)
	await frames(2)
	touch(0, false, cardc)
	await frames(5)
	log_lines.append("TOUCH_ONLY_CARD screen_after=%s" % c.screen)
	log_lines.append('post_card_levelups=%d screen=%d' % [await clear_levelups(), c.screen])
	await secs(1.0)
	# duraklat dokunma testi
	var pr: Vector2 = c.hud.pause_button.get_global_rect().get_center()
	touch(0, true, pr)
	await frames(2)
	touch(0, false, pr)
	await frames(5)
	log_lines.append("TOUCH_ONLY_PAUSE screen=%s" % c.screen)
	if c.screen != 3:
		c.pause_game()
	await secs(0.5)
	await shot("11-duraklatma")
	rects("pause", c.pause_screen)
	c.pause_screen.open_confirm(1)
	await secs(0.3)
	await shot("12-duraklatma-onay")
	rects("pause_confirm", c.pause_screen)
	c.pause_screen.close_confirm()
	c.resume_game()
	await secs(0.3)
	# oyun bitti
	c.sim.hp = 0.0
	c.sim.debug_spawn(&"sovalye_hayaleti", c.sim.player_pos)
	await secs(2.5)
	log_lines.append("after_death screen=%s state=%s" % [c.screen, c.sim.state])
	await shot("13-oyun-bitti-sayim")
	await secs(1.5)
	await shot("14-oyun-bitti")
	rects("result_lose", c.result_screen)
	# kazandın
	c.start_run()
	await secs(0.5)
	c.sim.debug_set_time(479.5)
	await secs(2.5)
	log_lines.append("after_win screen=%s state=%s" % [c.screen, c.sim.state])
	await secs(2.0)
	await shot("15-kazandin")
	rects("result_win", c.result_screen)
	# menü ses kapalı
	c.go_menu()
	c.menu.set_settings(false, false)
	await secs(0.5)
	await shot("16-menu-anahtar-kapali")
	var f := FileAccess.open(OUT + "log.txt", FileAccess.WRITE)
	f.store_string("\n".join(log_lines))
	f.close()
	quit()

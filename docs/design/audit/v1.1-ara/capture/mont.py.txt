import sys,os
STEP=int(os.environ.get("STEP","1"))
D=os.environ.get("D","o")
from PIL import Image
pre,n,cols,out=sys.argv[1],int(sys.argv[2]),int(sys.argv[3]),sys.argv[4]
fmt=sys.argv[5] if len(sys.argv)>5 else "%02d"
sc=float(sys.argv[6]) if len(sys.argv)>6 else 1
ims=[Image.open(f"{D}/{pre}_{fmt%(i*STEP)}.png").convert("RGB") for i in range(n)]
w,h=ims[0].size; w2,h2=int(w*sc),int(h*sc)
rows=(n+cols-1)//cols
s=Image.new("RGB",(w2*cols,h2*rows))
for i,im in enumerate(ims): s.paste(im.resize((w2,h2)),((i%cols)*w2,(i//cols)*h2))
s.save(out)
